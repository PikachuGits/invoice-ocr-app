### 2. OCR 客户端 (`ocr_client.rs`)

对应 Python 的 `submit_ocr` / `poll_result` / `download_image`：

```rust
use reqwest::Client;
use serde::{Deserialize, Serialize};
use std::time::Duration;
use thiserror::Error;

#[derive(Error, Debug)]
pub enum OcrError {
    #[error("HTTP error: {0}")]
    Http(#[from] reqwest::Error),
    #[error("OCR failed: {0}")]
    OcrFailed(String),
    #[error("Timeout after {0}s")]
    Timeout(u64),
}

#[derive(Serialize, Deserialize, Clone)]
pub struct OcrResult {
    pub html: String,
    pub images: Vec<String>, // base64 or URLs
}

pub struct OcrClient {
    client: Client,
    api_url: String,
}

impl OcrClient {
    pub fn new(api_url: &str) -> Self {
        let client = Client::builder()
            .timeout(Duration::from_secs(30))
            .build()
            .expect("Failed to build HTTP client");
        Self {
            client,
            api_url: api_url.to_string(),
        }
    }

    /// 提交图片进行 OCR，返回 task_id
    pub async fn submit(&self, image_path: &str) -> Result<String, OcrError> {
        let file_bytes = tokio::fs::read(image_path).await?;
        let part = reqwest::multipart::Part::bytes(file_bytes)
            .file_name("image.jpg")
            .mime_str("image/jpeg")?;
        let form = reqwest::multipart::Form::new().part("file", part);

        let resp = self.client.post(format!("{}/submit", self.api_url))
            .multipart(form)
            .send().await?;

        if !resp.status().is_success() {
            return Err(OcrError::OcrFailed(format!("Submit failed: {}", resp.status())));
        }

        let json: serde_json::Value = resp.json().await?;
        json["task_id"].as_str()
            .map(|s| s.to_string())
            .ok_or_else(|| OcrError::OcrFailed("No task_id in response".into()))
    }

    /// 轮询获取结果，支持超时和进度回调
    pub async fn poll<F>(
        &self,
        task_id: &str,
        timeout_secs: u64,
        on_progress: F,
    ) -> Result<OcrResult, OcrError>
    where
        F: Fn(f32) + Send + Sync,
    {
        let start = tokio::time::Instant::now();
        loop {
            if start.elapsed().as_secs() > timeout_secs {
                return Err(OcrError::Timeout(timeout_secs));
            }

            let resp = self.client.get(format!("{}/result/{}", self.api_url, task_id))
                .send().await?;
            let json: serde_json::Value = resp.json().await?;

            match json["status"].as_str() {
                Some("completed") => {
                    on_progress(1.0);
                    let html = json["html"].as_str().unwrap_or_default().to_string();
                    let images = json["images"]
                        .as_array()
                        .map(|arr| arr.iter().filter_map(|v| v.as_str().map(String::from)).collect())
                        .unwrap_or_default();
                    return Ok(OcrResult { html, images });
                }
                Some("processing") => {
                    let progress = json["progress"].as_f64().unwrap_or(0.0) as f32;
                    on_progress(progress);
                    tokio::time::sleep(Duration::from_millis(500)).await;
                }
                _ => {
                    return Err(OcrError::OcrFailed(format!("Unknown status: {:?}", json["status"])));
                }
            }
        }
    }

    /// 下载图片到本地路径
    pub async fn download_image(&self, url: &str, save_path: &str) -> Result<(), OcrError> {
        let resp = self.client.get(url).send().await?;
        let bytes = resp.bytes().await?;
        tokio::fs::write(save_path, bytes).await?;
        Ok(())
    }
}
```

### 3. HTML 表格解析 (`html_parser.rs`)

这是转换难点。Python 的 `_parse_span` + `occupied` 逻辑在 Rust 中需手动实现：

```rust
use scraper::{Html, Selector};
use std::collections::HashMap;

/// 解析后的单元格（处理了 rowspan/colspan）
#[derive(Debug, Clone)]
pub struct Cell {
    pub text: String,
    pub row: usize,
    pub col: usize,
    pub rowspan: usize,
    pub colspan: usize,
}

/// 将 HTML 表格解析为二维网格（已展开合并单元格）
pub fn parse_table(html: &str) -> Vec<Vec<String>> {
    let document = Html::parse_document(html);
    let table_sel = Selector::parse("table").unwrap();
    let tr_sel = Selector::parse("tr").unwrap();
    let td_sel = Selector::parse("td, th").unwrap();

    let mut grid: HashMap<(usize, usize), String> = HashMap::new();
    let mut max_row = 0usize;
    let mut max_col = 0usize;

    for (row_idx, tr) in document.select(&table_sel).flat_map(|t| t.select(&tr_sel)).enumerate() {
        let mut col_idx = 0usize;
        for td in tr.select(&td_sel) {
            // 跳过已被 rowspan/colspan 占据的位置
            while grid.contains_key(&(row_idx, col_idx)) {
                col_idx += 1;
            }

            let rowspan = td.attr("rowspan")
                .and_then(|v| v.parse::<usize>().ok())
                .unwrap_or(1);
            let colspan = td.attr("colspan")
                .and_then(|v| v.parse::<usize>().ok())
                .unwrap_or(1);
            let text = td.text().collect::<String>().trim().to_string();

            // 填充所有被合并覆盖的格子
            for r in 0..rowspan {
                for c in 0..colspan {
                    grid.insert((row_idx + r, col_idx + c), text.clone());
                }
            }

            max_row = max_row.max(row_idx + rowspan);
            max_col = max_col.max(col_idx + colspan);
            col_idx += colspan;
        }
    }

    // 转为二维 Vec
    let mut result = vec![vec![String::new(); max_col]; max_row];
    for ((r, c), text) in grid {
        if r < max_row && c < max_col {
            result[r][c] = text;
        }
    }
    result
}
```

> ⚠️ **注意**：`scraper` 不支持 `<tbody>` 隐式标签的完美解析，如果你的 OCR 返回 HTML 结构不规范，可能需要先用 `lol_html` 或正则预处理。建议保留一份原始 HTML 用于调试。

### 4. 发票字段提取 (`invoice_extractor.rs`)

直接移植 Python 的正则 + 关键字匹配逻辑：

```rust
use regex::Regex;
use serde::Serialize;

#[derive(Serialize, Clone)]
pub struct InvoiceData {
    pub buyer_name: Option<String>,
    pub seller_name: Option<String>,
    pub items: Vec<InvoiceItem>,
    pub total_amount: Option<f64>,
    pub tax_amount: Option<f64>,
    pub invoice_no: Option<String>,
    pub invoice_date: Option<String>,
}

#[derive(Serialize, Clone)]
pub struct InvoiceItem {
    pub name: String,
    pub quantity: Option<f64>,
    pub unit_price: Option<f64>,
    pub amount: Option<f64>,
    pub tax_rate: Option<f64>,
}

pub fn extract_invoice(grid: &[Vec<String>]) -> InvoiceData {
    let flat: Vec<String> = grid.iter().flatten().cloned().collect();
    let joined = flat.join("\n");

    let buyer_name = find_field(&joined, &["购买方", "购方名称", "买方"]);
    let seller_name = find_field(&joined, &["销售方", "销方名称", "卖方"]);
    let invoice_no = find_field(&joined, &["发票号码", "发票代码"]);
    let invoice_date = find_field(&joined, &["开票日期", "日期"]);

    let total_amount = find_number(&joined, &["合计金额", "价税合计", "总金额"]);
    let tax_amount = find_number(&joined, &["税额合计", "总税额"]);

    // 商品明细提取：查找包含"货物或应税劳务"等关键字的行作为表头，后续行为明细
    let items = extract_items(grid);

    InvoiceData {
        buyer_name,
        seller_name,
        items,
        total_amount,
        tax_amount,
        invoice_no,
        invoice_date,
    }
}

fn find_field(text: &str, keywords: &[&str]) -> Option<String> {
    for kw in keywords {
        if let Some(pos) = text.find(kw) {
            let after = &text[pos + kw.len()..];
            let line = after.lines().next().unwrap_or("").trim();
            if !line.is_empty() {
                return Some(line.to_string());
            }
        }
    }
    None
}

fn find_number(text: &str, keywords: &[&str]) -> Option<f64> {
    let re = Regex::new(r"[\d,]+\.?\d*").unwrap();
    for kw in keywords {
        if let Some(pos) = text.find(kw) {
            let after = &text[pos + kw.len()..];
            if let Some(m) = re.find(after) {
                let cleaned = m.as_str().replace(',', "");
                if let Ok(n) = cleaned.parse::<f64>() {
                    return Some(n);
                }
            }
        }
    }
    None
}

fn extract_items(grid: &[Vec<String>]) -> Vec<InvoiceItem> {
    // 简化版：假设第 N 行是表头，后续行是明细
    // 实际应根据关键字动态定位表头行
    let header_keywords = ["货物", "应税劳务", "项目名称", "规格型号"];
    let mut header_row = None;
    for (i, row) in grid.iter().enumerate() {
        if row.iter().any(|cell| header_keywords.iter().any(|kw| cell.contains(kw))) {
            header_row = Some(i);
            break;
        }
    }

    let mut items = Vec::new();
    if let Some(h) = header_row {
        for row in grid.iter().skip(h + 1) {
            if row.iter().all(|c| c.trim().is_empty()) { continue; }
            items.push(InvoiceItem {
                name: row.first().cloned().unwrap_or_default(),
                quantity: row.get(1).and_then(|s| s.replace(',', "").parse().ok()),
                unit_price: row.get(2).and_then(|s| s.replace(',', "").parse().ok()),
                amount: row.get(3).and_then(|s| s.replace(',', "").parse().ok()),
                tax_rate: row.get(4).and_then(|s| s.replace('%', "").replace(',', "").parse().ok()),
            });
        }
    }
    items
}
```

### 5. Tauri 命令 (`commands.rs`)

将上述模块暴露给前端调用：

```rust
use tauri::AppHandle;
use crate::ocr_client::OcrClient;
use crate::html_parser;
use crate::invoice_extractor;

#[tauri::command]
pub async fn recognize_invoice(app: AppHandle, image_path: String) -> Result<serde_json::Value, String> {
    let client = OcrClient::new("https://your-paddleocr-api.com");

    // 提交
    let task_id = client.submit(&image_path).await.map_err(|e| e.to_string())?;

    // 轮询（通过 app.emit 向前端推送进度）
    let result = client.poll(&task_id, 120, |progress| {
        let _ = app.emit("ocr-progress", progress);
    }).await.map_err(|e| e.to_string())?;

    // 解析 + 提取
    let grid = html_parser::parse_table(&result.html);
    let invoice = invoice_extractor::extract_invoice(&grid);

    Ok(serde_json::to_value(invoice).map_err(|e| e.to_string())?)
}
```

## 三、关键转换注意事项

| Python 原逻辑 | Rust 对应方案 | 风险点 |
|--------------|--------------|--------|
| `BeautifulSoup` 解析 | `scraper` crate | `scraper` 对不规范 HTML 容错性差，建议加 fallback 正则 |
| `requests` 同步请求 | `reqwest` + `tokio` 异步 | 必须全链路 async，不能混用阻塞调用 |
| `_parse_span` + `occupied` 集合 | `HashMap<(usize,usize), String>` | 逻辑等价，但需充分测试边界情况 |
| `re.findall` | `regex` crate | Rust regex 不支持回溯，复杂模式需改写 |
| `json.loads` | `serde_json::Value` | 动态 JSON 用 Value，结构化数据用 derive |
| `os.makedirs` | `tokio::fs::create_dir_all` | 异步版本，避免阻塞 runtime |
